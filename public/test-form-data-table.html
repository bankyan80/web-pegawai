KEP Form Data-Table Attribute Tester

Buka halaman /presensi di aplikasi, kemudian paste code di bawah di Console (F12 → Console):


// 1. Find all forms dengan class kep-form
var forms = document.querySelectorAll('form.kep-form');
console.log('Total kep-form found:', forms.length);

// 2. For each form, log data-table attribute
forms.forEach(function(form, idx) {
  var dataTable = form.getAttribute('data-table');
  var dataModal = form.getAttribute('data-modal');
  var batchWrap = form.querySelector('.kep-batch-wrap') ? 'YES' : 'NO';
  
  console.log('Form ' + idx + ':');
  console.log('  data-table: "' + dataTable + '"');
  console.log('  data-modal: "' + dataModal + '"');
  console.log('  has batch-wrap: ' + batchWrap);
  
  // Test tableModul if function exists
  if (typeof tableModul === 'function') {
    var modul = tableModul(dataTable);
    console.log('  tableModul("' + dataTable + '") → "' + modul + '"');
  }
  
  // If no data-table, check parent modal
  if (!dataTable) {
    var modal = form.closest('[role="dialog"]');
    if (modal) {
      console.log('  WARNING: Form dalam modal, modal id: ' + modal.id);
    }
  }
  console.log('---');
});

// 3. Check if tableModul is global
if (typeof tableModul === 'function') {
  console.log('✅ tableModul is global function');
} else {
  console.log('❌ tableModul is NOT global - cannot access from console');
}

// 4. Test all dynamic IDs
console.log('\nTesting tableModul patterns:');
var testIds = ['tblPrs0', 'tblPrs1', 'modPrs0', 'modPrs1', 'tblPresensi', 'tblPegawai', ''];
testIds.forEach(function(id) {
  if (typeof tableModul === 'function') {
    console.log('  tableModul("' + id + '") → "' + tableModul(id) + '"');
  }
});

Troubleshooting Steps:

  1. Buka halaman Presensi: /presensi
  2. Klik tombol "Upload PDF" pada salah satu tahun
  3. Modal akan terbuka
  4. Buka Developer Tools: F12
  5. Go ke tab Console
  6. Copy-paste code di atas
  7. Perhatikan output, terutama:
    • Apakah data-table attribute tidak kosong?
    • Apakah tableModul() return yang benar?
    • Apakah form punya has batch-wrap: YES?

Expected Output:

Total kep-form found: 1
Form 0:
  data-table: "tblPrs0"
  data-modal: "modPrs0"
  has batch-wrap: YES
  tableModul("tblPrs0") → "presensi"
✅ tableModul is global function

Testing tableModul patterns:
  tableModul("tblPrs0") → "presensi"
  tableModul("tblPrs1") → "presensi"
  tableModul("modPrs0") → "presensi"
  tableModul("modPrs1") → "presensi"
  tableModul("tblPresensi") → "presensi"
  tableModul("tblPegawai") → "pegawai"
  tableModul("") → ""

Jika output tidak sesuai dengan expected, screenshot dan kirim ke saya!
